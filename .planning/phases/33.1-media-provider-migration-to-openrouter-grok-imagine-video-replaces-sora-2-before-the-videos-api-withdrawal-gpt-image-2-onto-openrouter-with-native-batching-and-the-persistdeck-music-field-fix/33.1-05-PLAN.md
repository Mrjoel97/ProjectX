**The verified OpenRouter video contract** (PRD, confirmed 2026-08-30):

```
POST https://openrouter.ai/api/v1/videos                                 (JSON, not multipart)
Request  { model, prompt, duration?, resolution?, aspect_ratio?, size?,
           frame_images?, generate_audio?, seed?, callback_url? }
202      { id, polling_url, status: "pending" }
Poll     GET {polling_url}  ->  { id, status: pending|in_progress|completed|failed,
                                  unsigned_urls: [...], usage: { cost } }
Content  unsigned_urls[0]   or  GET /api/v1/videos/{id}/content?index=0
```

```ts
// buildSubmitBody, :915 — the video arm. Sora's field names.

  { jobId: line.jobId, videoId: res.requestId, attempt: 0 });

// pollOpenAiVideoTask, :1605 — the loop to mirror, verbatim in shape
export const pollOpenAiVideoTask = internalAction({
  args: { jobId: v.id("mediaJobs"), videoId: v.string(), attempt: v.number() },
  handler: async (ctx, a) => {
    const key = requireEnvMedia("OPENAI_API_KEY");
    const row = await ctx.runQuery(internal.media.jobForPoll, { jobId: a.jobId });
    if (!row || row.status !== "submitted" || row.spec.kind !== "video") return null;
    // fetch status; transport error -> reschedule to attempt+1 up to 180, else landResult
    //   poll_transport_error; !response.ok -> landResult providerReasonCode(response);
    //   status queued|in_progress -> reschedule (attempt >= 180 -> poll_timeout);
    //   status !== completed -> landResult with body.error.code through SAFE_CODE, else
    //     provider_failed; completed -> fetch content, landResult asset_transport_error /
    //     asset_http_N on failure, else:
    await storeAndLand(ctx, a.jobId, new Uint8Array(await asset.arrayBuffer()),
      asset.headers.get("content-type") ?? "video/mp4",
      { resolution: row.spec.resolution, seconds: row.spec.seconds });
  },
});
```
