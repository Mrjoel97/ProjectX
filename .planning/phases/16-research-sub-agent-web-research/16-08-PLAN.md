// __runSpecialistWithScript gained `softCutoffMs` (16-06 Task 1 step 6 → runSpecialistTurn's
// mockScript bag, 16-05 Task 2 step 3d). It is the ONLY way to drive the wall-clock row offline:
// mockScript crosses a Convex validator, so a "slow" mock cannot be a function, and asserting the
// real 180s clock would mean a 180-second test. Set it to 0 and the loop's SOFT stop condition
// (`elapsed >= softCutoffMs`) fires between steps on its FIRST evaluation, deterministically.
// It overrides the SOFT stop ONLY. There is deliberately no hard-budget override: shrinking
// `AbortSignal.timeout(...)` would make the abort race the mock and throw
// `ConvexError({kind:"agent_timeout"})` (llm.ts:1789) — the exact discard-the-work outcome this
// row exists to disprove, arriving as a green-looking failure.
const provUsage = (input, output) => ({ inputTokens: {total: input, …}, outputTokens: {…} });
const toolStep  = (toolName, input) => ({ content: [{type:"tool-call", toolCallId, toolName, input}],
                                          finishReason: {unified:"tool-calls"}, usage, warnings: [] });
const textStep  = (text, input = 0, output = 0) => ({ content: [{type:"text", text}], … });
const SPEND_8_CENTS = provUsage(100_000, 100_000);   // DEFAULT_MODEL $0.15/$0.60 per MTok
async function setup(): Promise<{ t: T; planId: Id<"plans"> }>   // seeds skills, registers
                        // the rateLimiter + auditCounts components, inserts a plan
