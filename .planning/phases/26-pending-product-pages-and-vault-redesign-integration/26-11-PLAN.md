```ts
export const promoteToReference = tenantMutation({
  args: { vaultDocId: v.id("vaultDocuments") },
  handler: async (ctx, { vaultDocId }) => {
    const doc = await ctx.db.get(vaultDocId);
    // Tenant term FIRST, and every foreign/missing outcome collapses to the SAME result — no
    // existence oracle. `tenantMutation` scopes nothing about `ctx.db.get`; this line is the isolation.
    if (!doc || doc.tenantId !== ctx.tenantId) return { ok: false, reason: "ineligible" } as const;
    // POSITIVE whitelist, never a negation: `origin` absent means USER-SUPPLIED (schema.ts:1665).
    if (doc.origin !== "agent" && doc.origin !== "agent_promoted")
      return { ok: false, reason: "ineligible" } as const;
    // Re-promotion is allowed for exactly one reason: a promoted row whose ingest FAILED is
    // otherwise unrecoverable (retryStuckIngests skips non-`processing` rows, patchCreatedDoc
    // refuses it). CAS on status here, not origin.
    if (doc.origin === "agent_promoted" && doc.status !== "failed")
      return { ok: true, state: "already_promoted" } as const;
    // ponytail: no version column, no retry loop, no index — a Convex mutation is a serializable
    // transaction, so the get+guard+patch above IS the compare-and-swap (cockpit.ts:975 precedent).
    // `status: "processing"` is NOT cosmetic: onIngestComplete early-returns on any row not in
    // "processing" (vaultIngest.ts:85), so without it a failed ingest is completely silent.
    await ctx.db.patch(vaultDocId, {
      origin: "agent_promoted",
      status: "processing",
      failureReason: undefined,
    });
    await startIngest(ctx, {
      vaultDocId,
      tenantId: ctx.tenantId,
      correlationId: `vault:promote:${vaultDocId}`,
    });
    // NO AUDIT ROW. **DEVIATION FROM THIS PLAN AS AUDITED, 2026-08-22 -- the plan was WRONG.**
    // The audited version told the executor to write `vault.promoted` from this mutation and
    // worried only about registering the auditCounts component. It missed vaultRedaction.test.ts,
    // a SHIPPED STATIC SCAN asserting that vault.ts / vaultIngest.ts / vaultGraph.ts / vaultLlm.ts
    // emit NO log-plane call or insert at all. That invariant is deliberate and STRONGER than the
    // payload assertion this plan proposed: vault.ts is the one module holding raw document text,
    // so it is kept log-free BY CONSTRUCTION -- a carefully-shaped payload is one careless edit

    // llm.ts, not by the vault plane. 26-13 owns the promotion audit when it builds the control.
    return { ok: true, state: "processing" } as const;
  },
});
```
