```ts
for (let i = 0; i < requestIds.length; i++) {
  const requestId = requestIds[i]; const cid = correlationIds[i];
  await step.runMutation(internal.pipeline.setStatus, { requestId, status: "delivering" });
  try {
    const result = await step.runAction(internal.gmail.send, { requestId }); // workpool retries — NOT retrier.run (no ctx in handler)
    if (!result.delivered) continue;                                         // held at awaiting_reauth — not a failure
    await step.runMutation(internal.pipeline.setStatus, { requestId, status: "sent" });
    await step.runMutation(internal.telemetry.writeTerminal, {
      requestId, correlationId: cid,
      outcome: { reviewOutcome: "sent", durationMs: 0, decisionCounts: {}, regenerateCount: 0, usages: [] }, // no LLM ran
    });
  } catch (e) {
    await step.runMutation(internal.deadLetter.deadLetterRecipient, { tenantId, requestId, correlationId: cid, error: String(e) });
  }
}
// mark the plan done at the end
await step.runMutation(internal.deliverApprovedPlan.markPlanDone, { planId }); // LOCAL setter (below) — no cross-plan dep so this plan compiles standalone in Wave 2
return null;
```
