```typescript
// packages/backend/convex/lib/functions.ts
import { customQuery, customMutation, customCtx } from "convex-helpers/server/customFunctions";
import { query, mutation } from "../_generated/server";   // THE ONLY allowed raw import
async function requireTenant(ctx: { auth: any }) {
  const identity = await ctx.auth.getUserIdentity();
  if (!identity) throw new Error("UNAUTHENTICATED");
  return identity.subject as string;                       // tenantId = userId for beta
}
export const tenantQuery = customQuery(query, customCtx(async (ctx) => ({ tenantId: await requireTenant(ctx) })));
export const tenantMutation = customMutation(mutation, customCtx(async (ctx) => ({ tenantId: await requireTenant(ctx) })));
```
