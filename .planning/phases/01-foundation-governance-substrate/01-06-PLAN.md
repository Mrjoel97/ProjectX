```typescript
// deadLetter.ts core
import { vWorkflowId, vResultValidator } from "@convex-dev/workflow";
export const onPipelineComplete = internalMutation({
  args: { workflowId: vWorkflowId, result: vResultValidator,
          context: v.object({ tenantId: v.string(), correlationId: v.string(), payload: v.any() }) },
  handler: async (ctx, { workflowId, result, context }) => {
    if (result.kind === "error" || result.kind === "canceled") {
      await ctx.db.insert("deadLetters", { tenantId: context.tenantId, correlationId: context.correlationId,
        workflowId, payload: context.payload, error: result.kind === "error" ? String(result.error) : "canceled",
        status: "new", createdAt: Date.now() });
      await ctx.runMutation(internal.audit.log, { /* deadletter.written event */ });
    } } });
```

Failing workflow = a step that throws with retry disabled ({ retry: false }).

awaitEvent race (issue #177 OPEN — workaround mandatory). Workflows cannot touch ctx.scheduler; arm inside a step:
```typescript
// inside workflow handler
await step.runMutation(internal.review.armTimeout, { workflowId, correlationId, timeoutMs });
const evt = await step.awaitEvent({ name: `review:${correlationId}`,
  validator: v.union(v.object({ kind: v.literal("decision"), decision: v.string() }),
                     v.object({ kind: v.literal("timeout") })) });
// review.ts: armTimeout -> scheduler.runAfter(timeoutMs, internal.review.fireTimeout, a) + insert pendingTimeouts
// fireTimeout -> sendEvent(ctx, components.workflow, { workflowId, name: `review:${correlationId}`, value: { kind: "timeout" } })
// sendDecision -> sends {kind:"decision"} AND ctx.scheduler.cancel(scheduledId) via pendingTimeouts lookup
```
Defensive rules (Pitfall 5): ALWAYS cancel scheduled timeout on real decision AND namespace event names by correlationId. Both.
