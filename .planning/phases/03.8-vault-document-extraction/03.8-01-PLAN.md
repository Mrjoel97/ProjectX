export type ExtractionKind = "pdf" | "image" | "office" | "transcribe";
/** mimeType-first; filename extension fallback (.docx/.xlsx/.pptx) because browsers
 *  sometimes send application/octet-stream for Office files. Returns null = not extractable
 *  (row stays pending_extraction / storage-only). */
export function extractionKindFor(mimeType: string, filename?: string): ExtractionKind | null;

// transcribe → video/* AND audio/* (classifier stays dumb — container acceptance is Lane 4's check)
export const VAULT_EXTRACT_CHAR_CAP = 400_000;  // chars stored via the seam (Pitfall 6: < 1 MiB doc cap)
